cmake_minimum_required(VERSION 3.16)
project(DPX_EXTERN VERSION 0.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(dpx_core INTERFACE)
target_include_directories(dpx_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
add_executable(dpx_demo src/main.cpp)
target_link_libraries(dpx_demo PRIVATE dpx_core)
add_executable(dpx_tests tests/tests.cpp)
target_link_libraries(dpx_tests PRIVATE dpx_core)
if(WIN32)
  add_executable(DPX-EXTERN WIN32 src/windows_gui.cpp)
  target_link_libraries(DPX-EXTERN PRIVATE dpx_core user32 gdi32)
endif()
enable_testing()
add_test(NAME core_tests COMMAND dpx_tests)
